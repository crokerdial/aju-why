async function handleUserMessage(text) { // 1. 사용자 메시지를 화면에 띄우고 대화 기록에 저장 renderMessage(text, 'user'); chatHistory.push({ "role": "user", "parts": [{ "text": text }] }); // 2. 시스템 프롬프트 (하브루타 코치 역할 및 JSON 형식 지정) const systemPrompt = `너는 4~5세 아이 부모를 돕는 '하브루타 대화 코치'야. JSON으로만 응답해. { "answer": "공감과 비유 설명", "question": "아이에게 던질 질문" }`; try { // 3. Vercel 백엔드 서버로 요청 보내기 const response = await fetch('/api/gemini', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ system_instruction: { parts: [{ text: systemPrompt }] }, contents: chatHistory }) }); const data = await response.json(); // 4. 에러가 반환되었을 경우 처리 if (data.error) { renderMessage("잠시 머리가 아파요! 🤯 (오류: " + data.error.message + ")", 'ai', false); return; } // 5. AI가 보낸 원본 텍스트 가져오기 const aiRaw = data.candidates[0].content.parts[0].text; // 🧹 6. [핵심] AI가 몰래 붙인 마크다운(```json) 찌꺼기 완벽 청소 const cleanText = aiRaw.replace(/```json/gi, '').replace(/```/g, '').trim(); // 7. 청소된 깨끗한 텍스트를 JSON 객체로 변환 const parsed = JSON.parse(cleanText); // 8. 화면에 AI의 공감 답변과 하브루타 질문 출력 // (UI에 더 예쁘게 나오도록 줄바꿈과 굵기 처리를 살짝 더했습니다!) renderMessage(parsed.answer + "

💡 " + parsed.question + "", 'ai'); // 9. 다음 대화를 위해 AI의 원본 답변을 기록에 저장 chatHistory.push({ "role": "model", "parts": [{ "text": aiRaw }] }); } catch (error) { console.error("프론트엔드 에러 상세 확인:", error); renderMessage("서버와 연결이 끊어졌어요. 기록 삭제 후 다시 시도해주세요.", 'ai', false); } }
